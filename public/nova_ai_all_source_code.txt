name: Build and Publish Debug APK

on:
  push:
    branches:
      - main
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Set up Java 17
        uses: actions/setup-java@v4
        with:
          distribution: 'temurin'
          java-version: '17'

      - name: Verify Java environment
        run: |
          java -version
          echo "JAVA_HOME=$JAVA_HOME"
          JAVA_MAJOR=$(java -version 2>&1 | awk -F '"' '/version/ {print $2}' | awk -F '.' '{print $1}')
          if [ -z "$JAVA_MAJOR" ]; then
            echo "::error::Could not detect active Java major version"
            exit 1
          fi
          echo "Detected Java major version: $JAVA_MAJOR"
          if [ "$JAVA_MAJOR" -lt 17 ]; then
            echo "::error::Active Java major version is $JAVA_MAJOR, but 17 or newer is required"
            exit 1
          fi

      - name: Set up Gradle 9.3.1
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '9.3.1'

      - name: Create temporary debug keystore
        run: |
          if [ -f "./debug.keystore" ]; then
            echo "debug.keystore already exists."
          else
            echo "Generating temporary debug.keystore..."
            keytool -genkey -v \
              -keystore ./debug.keystore \
              -storepass android \
              -alias androiddebugkey \
              -keypass android \
              -dname "CN=Android Debug,O=Android,C=US" \
              -keyalg RSA \
              -keysize 2048 \
              -validity 10000
          fi
          if [ ! -f "./debug.keystore" ]; then
            echo "::error::Failed to create ./debug.keystore"
            exit 1
          fi
          echo "./debug.keystore exists and is ready."

      - name: Build debug APK
        run: |
          gradle :app:assembleDebug --stacktrace --no-daemon

      - name: Verify the APK
        run: |
          EXPECTED_APK="app/build/outputs/apk/debug/app-debug.apk"
          if [ ! -f "$EXPECTED_APK" ]; then
            echo "::error::Expected APK $EXPECTED_APK does not exist!"
            echo "Searching inside app/build/outputs/apk:"
            find app/build/outputs/apk -type f -name "*.apk" || true
            exit 1
          fi

          if [ ! -s "$EXPECTED_APK" ]; then
            echo "::error::Expected APK $EXPECTED_APK is empty!"
            exit 1
          fi

          case "$EXPECTED_APK" in
            *.apk) ;;
            *) echo "::error::File does not have .apk extension!"; exit 1 ;;
          esac

          if [ "$EXPECTED_APK" != "app/build/outputs/apk/debug/app-debug.apk" ]; then
            echo "::error::Selected APK is not app/build/outputs/apk/debug/app-debug.apk"; exit 1
          fi

          echo "Verified APK at: $EXPECTED_APK"
          ls -la "$EXPECTED_APK"

      - name: Upload debug APK artifact
        uses: actions/upload-artifact@v4
        with:
          name: app-debug-apk
          path: app/build/outputs/apk/debug/app-debug.apk
          if-no-files-found: error

      - name: Prepare APK release asset
        id: prepare_asset
        run: |
          TEMP_DIR="${RUNNER_TEMP}/release-apk-asset"
          mkdir -p "$TEMP_DIR"
          REPO_RAW="${{ github.repository }}"
          REPO_BASE="${REPO_RAW##*/}"
          REPO_CLEAN=$(echo "$REPO_BASE" | tr '[:upper:]' '[:lower:]' | sed 's/[^a-z0-9]/-/g' | sed 's/-\{2,\}/-/g' | sed 's/^-//;s/-$//')
          if [ -z "$REPO_CLEAN" ]; then
            REPO_CLEAN="app"
          fi
          ASSET_FILENAME="${REPO_CLEAN}-debug-build-${{ github.run_number }}.apk"
          ASSET_PATH="${TEMP_DIR}/${ASSET_FILENAME}"
          cp app/build/outputs/apk/debug/app-debug.apk "$ASSET_PATH"

          if [ ! -s "$ASSET_PATH" ]; then
            echo "::error::Prepared release asset is missing or empty"
            exit 1
          fi

          echo "Prepared release asset: $ASSET_PATH"
          echo "asset_path=${ASSET_PATH}" >> "$GITHUB_OUTPUT"
          echo "asset_name=${ASSET_FILENAME}" >> "$GITHUB_OUTPUT"

      - name: Publish APK to GitHub Releases
        id: publish_release
        continue-on-error: true
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          gh --version
          ASSET_PATH="${{ steps.prepare_asset.outputs.asset_path }}"
          ASSET_NAME="${{ steps.prepare_asset.outputs.asset_name }}"
          if [ -z "$ASSET_PATH" ] || [ ! -f "$ASSET_PATH" ] || [ ! -s "$ASSET_PATH" ]; then
            echo "::error::Release asset validation failed before release creation."
            exit 1
          fi

          TAG="debug-apk-build-${{ github.run_number }}-${{ github.run_attempt }}"
          TITLE="Debug APK Build #${{ github.run_number }}"

          cat <<EOF > release_notes.md
          ### Debug APK Build
          This is an automatically generated debug APK.
          - **Built by:** GitHub Actions
          - **Run Number:** ${{ github.run_number }}
          - **Run Attempt:** ${{ github.run_attempt }}
          - **Source Commit:** ${{ github.sha }}

          > **Notice:** This APK is intended for testing. The APK is not a Play Store production release and is signed only with a temporary debug keystore.
          EOF

          echo "Creating release: $TAG with asset: $ASSET_PATH"
          gh release create "$TAG" "$ASSET_PATH" \
            --title "$TITLE" \
            --target "${{ github.sha }}" \
            --notes-file release_notes.md \
            --latest
          rm -f release_notes.md

      - name: Report APK download result
        if: always()
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          TAG="debug-apk-build-${{ github.run_number }}-${{ github.run_attempt }}"
          ASSET_NAME="${{ steps.prepare_asset.outputs.asset_name }}"
          if [ "${{ steps.publish_release.outcome }}" = "success" ]; then
            echo "GitHub Release publishing succeeded!"
            gh release view "$TAG" || true
            RELEASE_URL=$(gh release view "$TAG" --json url -q .url || echo "")
            echo "Release Tag: $TAG"
            echo "Release Title: Debug APK Build #${{ github.run_number }}"
            echo "Release URL: $RELEASE_URL"
            echo "Asset Filename: $ASSET_NAME"

            echo "### :rocket: Debug APK Release Created" >> "$GITHUB_STEP_SUMMARY"
            if [ -n "$RELEASE_URL" ]; then
              echo "- **Release Page:** [$RELEASE_URL]($RELEASE_URL)" >> "$GITHUB_STEP_SUMMARY"
            fi
            echo "- **Release Tag:** \`$TAG\`" >> "$GITHUB_STEP_SUMMARY"
            echo "- **Asset:** \`$ASSET_NAME\`" >> "$GITHUB_STEP_SUMMARY"
            echo "" >> "$GITHUB_STEP_SUMMARY"
            echo "The debug APK can be downloaded directly from the release Assets section on your phone." >> "$GITHUB_STEP_SUMMARY"
            echo "The normal Actions artifact (\`app-debug-apk\`) is also available as a backup." >> "$GITHUB_STEP_SUMMARY"
          else
            echo "::warning::GitHub Release publishing did not complete successfully or was skipped."
            echo "The APK build succeeded and the GitHub Actions artifact 'app-debug-apk' remains available for download."
            echo "### :warning: Release Publishing Notice" >> "$GITHUB_STEP_SUMMARY"
            echo "The debug APK build and artifact upload succeeded." >> "$GITHUB_STEP_SUMMARY"
            echo "GitHub Release publishing did not complete, but the debug APK is available under the **Artifacts** section as \`app-debug-apk\`." >> "$GITHUB_STEP_SUMMARY"
          fi

@Composable
fun NovaApp(
    viewModel: NovaViewModel = viewModel()
) {
    val isOnboardingDone by viewModel.isOnboardingCompleted.collectAsState()
    val voiceState by viewModel.voiceState.collectAsState()
    val errorMessage by viewModel.errorMessage.collectAsState()

    var currentScreen by remember {
        mutableStateOf(if (isOnboardingDone) NovaScreen.HOME else NovaScreen.WELCOME)
    }

    // Keep screen state synchronized with Voice State Machine
    LaunchedEffect(voiceState) {
        when (voiceState) {
            NovaVoiceState.LISTENING -> {
                if (currentScreen != NovaScreen.VOICE_LISTENING) {
                    currentScreen = NovaScreen.VOICE_LISTENING
                }
            }
            NovaVoiceState.PROCESSING -> {
                if (currentScreen != NovaScreen.THINKING) {
                    currentScreen = NovaScreen.THINKING
                }
            }
            NovaVoiceState.SPEAKING -> {
                if (currentScreen != NovaScreen.AI_RESPONSE && currentScreen != NovaScreen.INTERRUPT) {
                    currentScreen = NovaScreen.AI_RESPONSE
                }
            }
            NovaVoiceState.IDLE -> {
                if (currentScreen == NovaScreen.VOICE_LISTENING || currentScreen == NovaScreen.THINKING) {
                    currentScreen = NovaScreen.HOME
                }
            }
            NovaVoiceState.ERROR -> {
                if (currentScreen == NovaScreen.VOICE_LISTENING || currentScreen == NovaScreen.THINKING) {
                    currentScreen = NovaScreen.HOME
                }
            }
        }
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(NovaDarkBg)
    ) {
        AnimatedContent(
            targetState = currentScreen,
            transitionSpec = { fadeIn() togetherWith fadeOut() },
            label = "NovaScreenTransition"
        ) { screen ->
            when (screen) {
                NovaScreen.WELCOME -> {
                    WelcomeScreen(
                        onGetStarted = { currentScreen = NovaScreen.ONBOARDING }
                    )
                }

                NovaScreen.ONBOARDING -> {
                    BackHandler { currentScreen = NovaScreen.WELCOME }
                    OnboardingScreen(
                        onNext = {
                            if (viewModel.getEffectiveApiKey().isEmpty()) {
                                currentScreen = NovaScreen.API_KEY_SETUP
                            } else {
                                currentScreen = NovaScreen.MICROPHONE_PERMISSION
                            }
                        }
                    )
                }

                NovaScreen.API_KEY_SETUP -> {
                    BackHandler { currentScreen = NovaScreen.ONBOARDING }
                    ApiKeySetupScreen(
                        viewModel = viewModel,
                        onNavigateBack = { currentScreen = NovaScreen.ONBOARDING },
                        onSaveSuccess = {
                            currentScreen = NovaScreen.MICROPHONE_PERMISSION
                        }
                    )
                }

                NovaScreen.MICROPHONE_PERMISSION -> {
                    BackHandler {
                        viewModel.completeOnboarding()
                        currentScreen = NovaScreen.HOME
                    }
                    MicrophonePermissionScreen(
                        onPermissionGranted = {
                            viewModel.completeOnboarding()
                            currentScreen = NovaScreen.HOME
                        },
                        onSkip = {
                            viewModel.completeOnboarding()
                            currentScreen = NovaScreen.HOME
                        }
                    )
                }

                NovaScreen.HOME -> {
                    HomeScreen(
                        viewModel = viewModel,
                        onNavigateToSettings = { currentScreen = NovaScreen.SETTINGS },
                        onNavigateToHistory = { currentScreen = NovaScreen.CONVERSATION_HISTORY },
                        onNavigateToTextInput = { currentScreen = NovaScreen.TEXT_INPUT },
                        onStartVoiceListening = {
                            viewModel.startListening()
                            currentScreen = NovaScreen.VOICE_LISTENING
                        },
                        onQuickPromptSelected = { prompt ->
                            viewModel.submitQuestion(prompt)
                            currentScreen = NovaScreen.THINKING
                        }
                    )
                }

                NovaScreen.VOICE_LISTENING -> {
                    BackHandler {
                        viewModel.cancelListening()
                        currentScreen = NovaScreen.HOME
                    }
                    VoiceListeningScreen(
                        viewModel = viewModel,
                        onCancel = {
                            viewModel.cancelListening()
                            currentScreen = NovaScreen.HOME
                        },
                        onStop = {
                            val text = viewModel.speechRecognitionManager.partialText.value.trim()
                            if (text.isNotBlank()) {
                                viewModel.submitQuestion(text)
                                currentScreen = NovaScreen.THINKING
                            } else {
                                viewModel.stopListening()
                                currentScreen = NovaScreen.HOME
                            }
                        }
                    )
                }

                NovaScreen.THINKING -> {
                    BackHandler {
                        currentScreen = NovaScreen.HOME
                    }
                    ThinkingScreen(
                        viewModel = viewModel,
                        onCancel = {
                            currentScreen = NovaScreen.HOME
                        }
                    )
                }

                NovaScreen.AI_RESPONSE -> {
                    BackHandler {
                        viewModel.stopSpeaking()
                        currentScreen = NovaScreen.HOME
                    }
                    AiResponseScreen(
                        viewModel = viewModel,
                        onNavigateBack = {
                            viewModel.stopSpeaking()
                            currentScreen = NovaScreen.HOME
                        },
                        onNavigateToSettings = { currentScreen = NovaScreen.SETTINGS },
                        onAskAnother = {
                            viewModel.stopSpeaking()
                            viewModel.startListening()
                            currentScreen = NovaScreen.VOICE_LISTENING
                        }
                    )
                }

                NovaScreen.INTERRUPT -> {
                    BackHandler {
                        viewModel.stopSpeaking()
                        currentScreen = NovaScreen.HOME
                    }
                    InterruptScreen(
                        viewModel = viewModel,
                        onInterrupt = {
                            viewModel.stopSpeaking()
                            currentScreen = NovaScreen.HOME
                        }
                    )
                }

                NovaScreen.TEXT_INPUT -> {
                    BackHandler { currentScreen = NovaScreen.HOME }
                    TextInputScreen(
                        viewModel = viewModel,
                        onNavigateBack = { currentScreen = NovaScreen.HOME },
                        onNavigateToHome = { currentScreen = NovaScreen.HOME },
                        onNavigateToHistory = { currentScreen = NovaScreen.CONVERSATION_HISTORY },
                        onNavigateToSettings = { currentScreen = NovaScreen.SETTINGS },
                        onSubmitPrompt = { text ->
                            viewModel.submitQuestion(text)
                            currentScreen = NovaScreen.THINKING
                        }
                    )
                }

                NovaScreen.CONVERSATION_HISTORY -> {
                    BackHandler { currentScreen = NovaScreen.HOME }
                    HistoryScreen(
                        viewModel = viewModel,
                        onNavigateToHome = { currentScreen = NovaScreen.HOME },
                        onNavigateToSettings = { currentScreen = NovaScreen.SETTINGS },
                        onSelectConversation = { item ->
                            viewModel.submitQuestion(item.question)
                        }
                    )
                }

                NovaScreen.SETTINGS -> {
                    BackHandler { currentScreen = NovaScreen.HOME }
                    SettingsScreen(
                        viewModel = viewModel,
                        onNavigateToHome = { currentScreen = NovaScreen.HOME },
                        onNavigateToHistory = { currentScreen = NovaScreen.CONVERSATION_HISTORY },
                        onNavigateToLanguage = { currentScreen = NovaScreen.LANGUAGE_SELECTION },
                        onNavigateToVoice = { currentScreen = NovaScreen.VOICE_SPEECH },
                        onNavigateToApiConfig = { currentScreen = NovaScreen.API_CONFIG },
                        onNavigateToAbout = { currentScreen = NovaScreen.ABOUT }
                    )
                }

                NovaScreen.LANGUAGE_SELECTION -> {
                    BackHandler { currentScreen = NovaScreen.SETTINGS }
                    LanguageSelectionScreen(
                        viewModel = viewModel,
                        onNavigateBack = { currentScreen = NovaScreen.SETTINGS }
                    )
                }

                NovaScreen.VOICE_SPEECH -> {
                    BackHandler { currentScreen = NovaScreen.SETTINGS }
                    VoiceSpeechScreen(
                        viewModel = viewModel,
                        onNavigateBack = { currentScreen = NovaScreen.SETTINGS }
                    )
                }

                NovaScreen.API_CONFIG -> {
                    BackHandler { currentScreen = NovaScreen.SETTINGS }
                    ApiConfigScreen(
                        viewModel = viewModel,
                        onNavigateBack = { currentScreen = NovaScreen.SETTINGS },
                        onEditKey = { currentScreen = NovaScreen.API_KEY_SETUP }
                    )
                }

                NovaScreen.ABOUT -> {
                    BackHandler { currentScreen = NovaScreen.SETTINGS }
                    AboutScreen(
                        onNavigateBack = { currentScreen = NovaScreen.SETTINGS }
                    )
                }
            }
        }

        // Error Dialog
        if (errorMessage != null) {
            AlertDialog(
                onDismissRequest = { viewModel.dismissError() },
                title = {
                    Text(
                        text = "NOVA AI Notice",
                        color = NovaTextPrimary,
                        fontWeight = FontWeight.Bold
                    )
                },
                text = {
                    Text(
                        text = errorMessage ?: "",
                        color = NovaTextSecondary,
                        fontSize = 14.sp
                    )
                },
                confirmButton = {
                    TextButton(onClick = { viewModel.dismissError() }) {
                        Text("OK", color = NovaCyanBright)
                    }
                },
                dismissButton = {
                    if (voiceState == NovaVoiceState.ERROR) {
                        TextButton(
                            onClick = {
                                viewModel.dismissError()
                                viewModel.retryLastQuestion()
                            }
                        ) {
                            Text("Retry", color = NovaCyanBright)
                        }
                    }
                },
                containerColor = NovaDarkSurfaceVariant,
                shape = RoundedCornerShape(20.dp)
            )
        }
    }
}

package com.example.ui.viewmodel

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import com.example.BuildConfig
import com.example.data.local.AppDatabase
import com.example.data.local.ConversationEntity
import com.example.data.local.PreferencesManager
import com.example.data.network.gemini.GeminiClient
import com.example.speech.NovaVoiceOption
import com.example.speech.SpeechRecognitionManager
import com.example.speech.TextToSpeechManager
import kotlinx.coroutines.Job
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharingStarted
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.stateIn
import kotlinx.coroutines.launch

class NovaViewModel(application: Application) : AndroidViewModel(application) {

    private val db = AppDatabase.getInstance(application)
    private val conversationDao = db.conversationDao()
    private val preferencesManager = PreferencesManager(application)
    private val geminiClient = GeminiClient()

    val speechRecognitionManager = SpeechRecognitionManager(application)
    val textToSpeechManager = TextToSpeechManager(application)

    private val _voiceState = MutableStateFlow(NovaVoiceState.IDLE)
    val voiceState: StateFlow<NovaVoiceState> = _voiceState.asStateFlow()

    private val _currentQuestion = MutableStateFlow("")
    val currentQuestion: StateFlow<String> = _currentQuestion.asStateFlow()

    private val _currentAnswer = MutableStateFlow("")
    val currentAnswer: StateFlow<String> = _currentAnswer.asStateFlow()

    private val _errorMessage = MutableStateFlow<String?>(null)
    val errorMessage: StateFlow<String?> = _errorMessage.asStateFlow()

    private val _testConnectionLoading = MutableStateFlow(false)
    val testConnectionLoading: StateFlow<Boolean> = _testConnectionLoading.asStateFlow()

    private val _isApiConnected = MutableStateFlow<Boolean?>(null)
    val isApiConnected: StateFlow<Boolean?> = _isApiConnected.asStateFlow()

    private val _testConnectionMessage = MutableStateFlow<String?>(null)
    val testConnectionMessage: StateFlow<String?> = _testConnectionMessage.asStateFlow()

    val isOnboardingCompleted: StateFlow<Boolean> = preferencesManager.isOnboardingCompleted
        .stateIn(viewModelScope, SharingStarted.Eagerly, false)

    val savedApiKey: StateFlow<String> = preferencesManager.apiKey
        .stateIn(viewModelScope, SharingStarted.Eagerly, "")

    val preferredLanguage: StateFlow<String> = preferencesManager.preferredLanguage
        .stateIn(viewModelScope, SharingStarted.Eagerly, "bn")

    val speechSpeed: StateFlow<Float> = preferencesManager.speechSpeed
        .stateIn(viewModelScope, SharingStarted.Eagerly, 1.0f)

    val selectedVoice: StateFlow<String> = preferencesManager.selectedVoice
        .stateIn(viewModelScope, SharingStarted.Eagerly, "")

    val allConversations: StateFlow<List<ConversationEntity>> = conversationDao.getAllConversations()
        .stateIn(viewModelScope, SharingStarted.WhileSubscribed(5000), emptyList())

    val availableVoices: StateFlow<List<NovaVoiceOption>> = textToSpeechManager.availableVoices

    private var currentProcessingJob: Job? = null

    init {
        // Collect TTS and speech recognition states to keep UI synchronized
        viewModelScope.launch {
            speechRecognitionManager.errorMessage.collect { err ->
                if (err != null && _voiceState.value == NovaVoiceState.LISTENING) {
                    _voiceState.value = NovaVoiceState.IDLE
                    _errorMessage.value = err
                }
            }
        }
    }

    fun getEffectiveApiKey(): String {
        val userKey = savedApiKey.value.trim()
        if (userKey.isNotEmpty()) return userKey

        val buildKey = try {
            BuildConfig.GEMINI_API_KEY.trim()
        } catch (_: Exception) {
            ""
        }
        if (buildKey.isNotEmpty() && !buildKey.startsWith("MY_GEMINI_API_KEY")) {
            return buildKey
        }
        return ""
    }

    fun startListening() {
        if (_voiceState.value == NovaVoiceState.SPEAKING) {
            stopSpeaking()
        }
        _voiceState.value = NovaVoiceState.LISTENING
        _errorMessage.value = null

        speechRecognitionManager.startListening(preferredLanguage.value) { resultText ->
            if (resultText.isNotBlank()) {
                submitQuestion(resultText)
            } else {
                _voiceState.value = NovaVoiceState.IDLE
            }
        }
    }

    fun stopListening() {
        val text = speechRecognitionManager.partialText.value.trim()
        if (text.isNotBlank()) {
            submitQuestion(text)
        } else {
            speechRecognitionManager.cancelListening()
            _voiceState.value = NovaVoiceState.IDLE
        }
    }

    fun cancelListening() {
        speechRecognitionManager.cancelListening()
        _voiceState.value = NovaVoiceState.IDLE
    }

    fun submitQuestion(prompt: String) {
        val query = prompt.trim()
        if (query.isEmpty()) return

        speechRecognitionManager.stopListening()
        if (_voiceState.value == NovaVoiceState.SPEAKING) {
            stopSpeaking()
        }

        _currentQuestion.value = query
        _currentAnswer.value = ""
        _voiceState.value = NovaVoiceState.PROCESSING
        _errorMessage.value = null

        val apiKey = getEffectiveApiKey()
        if (apiKey.isEmpty()) {
            _voiceState.value = NovaVoiceState.ERROR
            _errorMessage.value = "Gemini API key is not configured. Please add your API key in Settings or Setup screen."
            return
        }

        currentProcessingJob?.cancel()
        currentProcessingJob = viewModelScope.launch {
            val contextPairs = allConversations.value.take(4).map { it.question to it.answer }
            val result = geminiClient.askGemini(
                prompt = query,
                apiKey = apiKey,
                preferredLanguage = preferredLanguage.value,
                contextHistory = contextPairs
            )

            result.onSuccess { answer ->
                _currentAnswer.value = answer
                _voiceState.value = NovaVoiceState.SPEAKING

                // Save to local database
                conversationDao.insertConversation(
                    ConversationEntity(
                        question = query,
                        answer = answer,
                        languageCode = preferredLanguage.value
                    )
                )

                // Speak response
                textToSpeechManager.speak(
                    text = answer,
                    languageCode = preferredLanguage.value,
                    speechRate = speechSpeed.value,
                    voiceName = selectedVoice.value.ifBlank { null }
                ) {
                    if (_voiceState.value == NovaVoiceState.SPEAKING) {
                        _voiceState.value = NovaVoiceState.IDLE
                    }
                }
            }.onFailure { ex ->
                _voiceState.value = NovaVoiceState.ERROR
                _errorMessage.value = ex.message ?: "Failed to get AI response. Please try again."
            }
        }
    }

    fun stopSpeaking() {
        textToSpeechManager.stop()
        if (_voiceState.value == NovaVoiceState.SPEAKING) {
            _voiceState.value = NovaVoiceState.IDLE
        }
    }

    fun speakCurrentAnswer() {
        val text = _currentAnswer.value
        if (text.isNotBlank()) {
            _voiceState.value = NovaVoiceState.SPEAKING
            textToSpeechManager.speak(
                text = text,
                languageCode = preferredLanguage.value,
                speechRate = speechSpeed.value,
                voiceName = selectedVoice.value.ifBlank { null }
            ) {
                if (_voiceState.value == NovaVoiceState.SPEAKING) {
                    _voiceState.value = NovaVoiceState.IDLE
                }
            }
        }
    }

    fun retryLastQuestion() {
        val lastQuestion = _currentQuestion.value
        if (lastQuestion.isNotBlank()) {
            submitQuestion(lastQuestion)
        }
    }

    fun dismissError() {
        _errorMessage.value = null
        _voiceState.value = NovaVoiceState.IDLE
    }

    fun testApiConnection(key: String, onFinished: ((Boolean, String) -> Unit)? = null) {
        val trimmed = key.trim()
        if (trimmed.isEmpty()) {
            _isApiConnected.value = false
            _testConnectionMessage.value = "Please enter a valid API key."
            onFinished?.invoke(false, "Please enter a valid API key.")
            return
        }

        _testConnectionLoading.value = true
        _testConnectionMessage.value = null

        viewModelScope.launch {
            val result = geminiClient.testConnection(trimmed)
            _testConnectionLoading.value = false
            result.onSuccess {
                _isApiConnected.value = true
                val msg = "Connection successful! Your API key is valid."
                _testConnectionMessage.value = msg
                onFinished?.invoke(true, msg)
            }.onFailure { ex ->
                _isApiConnected.value = false
                val msg = ex.message ?: "Connection test failed."
                _testConnectionMessage.value = msg
                onFinished?.invoke(false, msg)
            }
        }
    }

    fun saveApiKey(key: String) {
        viewModelScope.launch {
            preferencesManager.setApiKey(key.trim())
            _isApiConnected.value = true
        }
    }

    fun deleteApiKey() {
        viewModelScope.launch {
            preferencesManager.clearApiKey()
            _isApiConnected.value = null
            _testConnectionMessage.value = null
        }
    }

    fun savePreferredLanguage(lang: String) {
        viewModelScope.launch {
            preferencesManager.setPreferredLanguage(lang)
        }
    }

    fun saveSpeechSpeed(speed: Float) {
        viewModelScope.launch {
            preferencesManager.setSpeechSpeed(speed)
        }
    }

    fun saveSelectedVoice(voiceName: String) {
        viewModelScope.launch {
            preferencesManager.setSelectedVoice(voiceName)
        }
    }

    fun completeOnboarding() {
        viewModelScope.launch {
            preferencesManager.setOnboardingCompleted(true)
        }
    }

    fun deleteConversation(id: Long) {
        viewModelScope.launch {
            conversationDao.deleteConversationById(id)
        }
    }

    fun clearAllConversations() {
        viewModelScope.launch {
            conversationDao.clearAllConversations()
        }
    }

    fun testSampleVoice(voiceOption: NovaVoiceOption) {

import com.squareup.moshi.Moshi
import com.squareup.moshi.kotlin.reflect.KotlinJsonAdapterFactory
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import okhttp3.MediaType.Companion.toMediaType
import okhttp3.OkHttpClient
import okhttp3.Request
import okhttp3.RequestBody.Companion.toRequestBody
import java.io.IOException
import java.util.concurrent.TimeUnit

class GeminiClient {

    private val moshi = Moshi.Builder()
        .add(KotlinJsonAdapterFactory())
        .build()

    private val requestAdapter = moshi.adapter(GeminiRequest::class.java)
    private val responseAdapter = moshi.adapter(GeminiResponse::class.java)
    private val listModelsAdapter = moshi.adapter(ListModelsResponse::class.java)

    private val okHttpClient = OkHttpClient.Builder()
        .connectTimeout(20, TimeUnit.SECONDS)
        .readTimeout(30, TimeUnit.SECONDS)
        .writeTimeout(20, TimeUnit.SECONDS)
        .build()

    companion object {
        private val TEST_ENDPOINTS = listOf(
            "https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent",
            "https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent",
            "https://generativelanguage.googleapis.com/v1beta/models/gemini-3.5-flash:generateContent",
            "https://generativelanguage.googleapis.com/v1/models/gemini-2.0-flash:generateContent",
            "https://generativelanguage.googleapis.com/v1/models/gemini-1.5-flash:generateContent"
        )
        private val JSON_MEDIA_TYPE = "application/json; charset=utf-8".toMediaType()

        @Volatile
        private var cachedWorkingEndpoint: String? = null
    }

    private fun cleanKey(key: String): String {
        return key.trim().trim('"', '\'', ' ', '\n', '\r')
    }

    suspend fun testConnection(apiKey: String): Result<Boolean> = withContext(Dispatchers.IO) {
        val trimmedKey = cleanKey(apiKey)
        if (trimmedKey.isEmpty()) {
            return@withContext Result.failure(IllegalArgumentException("API Key cannot be empty."))
        }

        // 1. Try to discover supported models via ListModels (non-fatal if restricted)
        val discoveredEndpoints = mutableListOf<String>()
        try {
            val listUrl = "https://generativelanguage.googleapis.com/v1beta/models?key=$trimmedKey"
            val listReq = Request.Builder()
                .url(listUrl)
                .header("x-goog-api-key", trimmedKey)
                .get()
                .build()

            okHttpClient.newCall(listReq).execute().use { response ->
                if (response.isSuccessful) {
                    val bodyString = response.body?.string() ?: ""
                    val listResponse = listModelsAdapter.fromJson(bodyString)
                    val supported = listResponse?.models?.filter {
                        it.supportedGenerationMethods?.contains("generateContent") == true
                    }?.map { it.name.removePrefix("models/") }

                    if (!supported.isNullOrEmpty()) {
                        // Prefer 2.0-flash, 2.5-flash, or any flash model
                        val sorted = supported.sortedByDescending {
                            when {
                                it.contains("2.0-flash") -> 100
                                it.contains("2.5-flash") -> 90
                                it.contains("3.5-flash") -> 80
                                it.contains("flash") -> 70
                                else -> 10
                            }
                        }
                        for (model in sorted) {
                            discoveredEndpoints.add("https://generativelanguage.googleapis.com/v1beta/models/$model:generateContent")
                        }
                    }
                }
            }
        } catch (_: Exception) {}

        // Combine discovered endpoints with known reliable endpoints
        val candidateEndpoints = (discoveredEndpoints + TEST_ENDPOINTS).distinct()

        val testPayload = GeminiRequest(
            contents = listOf(
                ContentItem(
                    role = "user",
                    parts = listOf(PartItem(text = "Hello, reply with OK."))
                )
            )
        )
        val jsonBody = requestAdapter.toJson(testPayload)
        val body = jsonBody.toRequestBody(JSON_MEDIA_TYPE)

        var lastErrorMsg: String? = null

        for (endpoint in candidateEndpoints) {
            val fullUrl = "$endpoint?key=$trimmedKey"
            val request = Request.Builder()
                .url(fullUrl)
                .header("x-goog-api-key", trimmedKey)
                .post(body)
                .build()

            try {
                okHttpClient.newCall(request).execute().use { response ->
                    val responseString = response.body?.string() ?: ""
                    if (response.isSuccessful) {
                        val parsed = responseAdapter.fromJson(responseString)
                        val text = parsed?.candidates?.firstOrNull()?.content?.parts?.firstOrNull()?.text
                        if (!text.isNullOrBlank()) {
                            cachedWorkingEndpoint = endpoint
                            return@withContext Result.success(true)
                        }
                    } else if (response.code == 404) {
                        // Model not supported on this endpoint, continue trying others
                        lastErrorMsg = "Model not found on endpoint. Trying alternatives..."
                    } else {
                        val parsedError = try {
                            responseAdapter.fromJson(responseString)?.error?.message
                        } catch (_: Exception) {
                            null
                        }
                        lastErrorMsg = when (response.code) {
                            400 -> parsedError ?: "API key is invalid. Please check your key in Google AI Studio."
                            403 -> parsedError ?: "API key lacks permission or Gemini API is not enabled in Google Cloud."
                            429 -> "API quota exceeded. Please check Google AI Studio."
                            else -> parsedError ?: "HTTP ${response.code}: Connection failed."
                        }
                        if (response.code == 400 || response.code == 403) {
                            return@withContext Result.failure(IOException(lastErrorMsg))
                        }
                    }
                }
            } catch (e: Exception) {
                lastErrorMsg = e.message
            }
        }

        Result.failure(IOException(lastErrorMsg ?: "Could not connect to Gemini API. Please verify your internet and API key."))
    }

    suspend fun askGemini(
        prompt: String,
        apiKey: String,
        preferredLanguage: String,
        contextHistory: List<Pair<String, String>> = emptyList()
    ): Result<String> = withContext(Dispatchers.IO) {
        val trimmedKey = cleanKey(apiKey)
        if (trimmedKey.isEmpty()) {
            return@withContext Result.failure(IllegalStateException("Gemini API key is not configured."))
        }

        val languageInstruction = when (preferredLanguage.lowercase()) {
            "bn", "bengali" -> "You are NOVA AI, an intelligent, helpful, and friendly voice personal assistant. The user prefers Bengali (বাংলা). Answer clearly, conversationally, and accurately in Bengali. Keep answers concise and suitable for spoken text."
            "hi", "hindi" -> "You are NOVA AI, an intelligent, helpful, and friendly voice personal assistant. The user prefers Hindi (हिन्दी). Answer clearly, conversationally, and accurately in Hindi. Keep answers concise and suitable for spoken text."
            "ar", "arabic" -> "You are NOVA AI, an intelligent, helpful, and friendly voice personal assistant. The user prefers Arabic. Answer clearly, conversationally, and accurately in Arabic. Keep answers concise and suitable for spoken text."
            "es", "spanish" -> "You are NOVA AI, an intelligent, helpful, and friendly voice personal assistant. The user prefers Spanish. Answer clearly, conversationally, and accurately in Spanish. Keep answers concise and suitable for spoken text."
            else -> "You are NOVA AI, an intelligent, helpful, and friendly voice personal assistant. Answer clearly, conversationally, and accurately in English. Keep answers concise, natural, and suitable for spoken text."
        }

        val contentsList = mutableListOf<ContentItem>()

        for (item in contextHistory.takeLast(3)) {
            contentsList.add(ContentItem(role = "user", parts = listOf(PartItem(text = item.first))))
            contentsList.add(ContentItem(role = "model", parts = listOf(PartItem(text = item.second))))
        }

        contentsList.add(ContentItem(role = "user", parts = listOf(PartItem(text = prompt))))

        val geminiRequest = GeminiRequest(
            contents = contentsList,
            systemInstruction = ContentItem(
                parts = listOf(PartItem(text = languageInstruction))
            )
        )

        val jsonBody = requestAdapter.toJson(geminiRequest)
        val body = jsonBody.toRequestBody(JSON_MEDIA_TYPE)

        val baseEndpoint = cachedWorkingEndpoint ?: TEST_ENDPOINTS.first()
        val candidateEndpoints = listOf(baseEndpoint) + TEST_ENDPOINTS.filter { it != baseEndpoint }

        for (endpoint in candidateEndpoints) {
            val fullUrl = "$endpoint?key=$trimmedKey"
            val request = Request.Builder()
                .url(fullUrl)
                .header("x-goog-api-key", trimmedKey)
                .post(body)
                .build()

            try {
                okHttpClient.newCall(request).execute().use { response ->
                    val responseString = response.body?.string() ?: ""
                    if (response.isSuccessful) {
                        val parsed = responseAdapter.fromJson(responseString)
                        val text = parsed?.candidates?.firstOrNull()?.content?.parts?.firstOrNull()?.text
                        if (!text.isNullOrBlank()) {
                            cachedWorkingEndpoint = endpoint
                            return@withContext Result.success(text.trim())
                        } else {
                            return@withContext Result.failure(IOException("Empty response received from Gemini."))
                        }
                    } else if (response.code == 404 && endpoint != candidateEndpoints.last()) {
                        return@use
                    } else {
                        val errorDetails = try {
                            responseAdapter.fromJson(responseString)?.error?.message
                        } catch (_: Exception) {
                            null
                        }
                        val msg = when (response.code) {
                            400 -> errorDetails ?: "Invalid request format."
                            403 -> errorDetails ?: "Invalid API key or unauthorized."
                            429 -> "Gemini rate limit exceeded. Please wait a moment."
                            else -> errorDetails ?: "HTTP ${response.code}: Failed to get response."
                        }
                        return@withContext Result.failure(IOException(msg))
                    }
                }
            } catch (e: Exception) {
                if (endpoint == candidateEndpoints.last()) {
                    return@withContext Result.failure(e)
                }
            }
        }

        Result.failure(IOException("Network error communicating with Gemini."))
    }
}

package com.example.speech

import android.content.Context
import android.content.Intent
import android.os.Bundle
import android.os.Handler
import android.os.Looper
import android.speech.RecognitionListener
import android.speech.RecognizerIntent
import android.speech.SpeechRecognizer
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import java.util.Locale

class SpeechRecognitionManager(private val context: Context) {

    private var speechRecognizer: SpeechRecognizer? = null
    private val mainHandler = Handler(Looper.getMainLooper())

    private val _isListening = MutableStateFlow(false)
    val isListening: StateFlow<Boolean> = _isListening.asStateFlow()

    private val _partialText = MutableStateFlow("")
    val partialText: StateFlow<String> = _partialText.asStateFlow()

    private val _rmsLevel = MutableStateFlow(0f)
    val rmsLevel: StateFlow<Float> = _rmsLevel.asStateFlow()

    private val _errorMessage = MutableStateFlow<String?>(null)
    val errorMessage: StateFlow<String?> = _errorMessage.asStateFlow()

    private var onSpeechResultListener: ((String) -> Unit)? = null

    init {
        mainHandler.post {
            initRecognizer()
        }
    }

    private fun initRecognizer() {
        if (SpeechRecognizer.isRecognitionAvailable(context)) {
            speechRecognizer = SpeechRecognizer.createSpeechRecognizer(context).apply {
                setRecognitionListener(createListener())
            }
        }
    }

    private fun createListener(): RecognitionListener {
        return object : RecognitionListener {
            override fun onReadyForSpeech(params: Bundle?) {
                _isListening.value = true
                _errorMessage.value = null
            }

            override fun onBeginningOfSpeech() {
                _isListening.value = true
            }

            override fun onRmsChanged(rmsdB: Float) {
                // Normalize dB (typically -2 to 10) to 0.0 .. 1.0 range
                val normalized = ((rmsdB + 2f) / 12f).coerceIn(0f, 1f)
                _rmsLevel.value = normalized
            }

            override fun onBufferReceived(buffer: ByteArray?) {}

            override fun onEndOfSpeech() {
                _isListening.value = false
            }

            override fun onError(error: Int) {
                _isListening.value = false
                _rmsLevel.value = 0f
                val message = when (error) {
                    SpeechRecognizer.ERROR_AUDIO -> "Audio recording error"
                    SpeechRecognizer.ERROR_CLIENT -> "Speech recognition client error"
                    SpeechRecognizer.ERROR_INSUFFICIENT_PERMISSIONS -> "Microphone permission required"
                    SpeechRecognizer.ERROR_NETWORK -> "Network connection error"
                    SpeechRecognizer.ERROR_NETWORK_TIMEOUT -> "Network timed out"
                    SpeechRecognizer.ERROR_NO_MATCH -> "No speech detected. Please speak clearly."
                    SpeechRecognizer.ERROR_RECOGNIZER_BUSY -> "Recognition service is busy"
                    SpeechRecognizer.ERROR_SERVER -> "Recognition server error"
                    SpeechRecognizer.ERROR_SPEECH_TIMEOUT -> "No speech input received"
                    else -> "Speech recognition encountered an issue"
                }
                _errorMessage.value = message
            }

            override fun onResults(results: Bundle?) {
                _isListening.value = false
                _rmsLevel.value = 0f
                val matches = results?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)
                val recognized = matches?.firstOrNull() ?: ""
                _partialText.value = recognized
                if (recognized.isNotBlank()) {
                    onSpeechResultListener?.invoke(recognized)
                }
            }

            override fun onPartialResults(partialResults: Bundle?) {
                val matches = partialResults?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)
                val text = matches?.firstOrNull() ?: ""
                if (text.isNotBlank()) {
                    _partialText.value = text
                }
            }

            override fun onEvent(eventType: Int, params: Bundle?) {}
        }
    }

    fun startListening(languageCode: String, onResult: (String) -> Unit) {
        onSpeechResultListener = onResult
        _partialText.value = ""
        _errorMessage.value = null

        mainHandler.post {
            try {
                if (speechRecognizer == null) {
                    initRecognizer()
                }

                val localeTag = when (languageCode.lowercase()) {
                    "bn", "bengali" -> "bn-BD"
                    "hi", "hindi" -> "hi-IN"
                    "ar", "arabic" -> "ar-SA"
                    "es", "spanish" -> "es-ES"
                    else -> "en-US"
                }

                val intent = Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH).apply {
                    putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM)
                    putExtra(RecognizerIntent.EXTRA_LANGUAGE, localeTag)
                    putExtra(RecognizerIntent.EXTRA_LANGUAGE_PREFERENCE, localeTag)
                    putExtra(RecognizerIntent.EXTRA_ONLY_RETURN_LANGUAGE_PREFERENCE, localeTag)
                    putExtra(RecognizerIntent.EXTRA_PARTIAL_RESULTS, true)
                    putExtra(RecognizerIntent.EXTRA_MAX_RESULTS, 3)
                }

                speechRecognizer?.startListening(intent)
                _isListening.value = true
            } catch (e: Exception) {
                _isListening.value = false
                _errorMessage.value = "Failed to start speech recognition: ${e.message}"
            }
        }
    }

    fun stopListening() {
        mainHandler.post {
            try {
                speechRecognizer?.stopListening()
            } catch (_: Exception) {}
            _isListening.value = false
            _rmsLevel.value = 0f
        }
    }

    fun cancelListening() {
        mainHandler.post {
            try {
                speechRecognizer?.cancel()
            } catch (_: Exception) {}
            _isListening.value = false
            _partialText.value = ""
            _rmsLevel.value = 0f
        }
    }

    fun isAvailable(): Boolean {
        return SpeechRecognizer.isRecognitionAvailable(context)
    }

    fun destroy() {
        mainHandler.post {
            try {
                speechRecognizer?.destroy()
                speechRecognizer = null
            } catch (_: Exception) {}
        }
    }
}

class TextToSpeechManager(private val context: Context) : TextToSpeech.OnInitListener {

    private var tts: TextToSpeech? = null

    private val _isReady = MutableStateFlow(false)
    val isReady: StateFlow<Boolean> = _isReady.asStateFlow()

    private val _isSpeaking = MutableStateFlow(false)
    val isSpeaking: StateFlow<Boolean> = _isSpeaking.asStateFlow()

    private val _availableVoices = MutableStateFlow<List<NovaVoiceOption>>(emptyList())
    val availableVoices: StateFlow<List<NovaVoiceOption>> = _availableVoices.asStateFlow()

    private var onSpeechDoneCallback: (() -> Unit)? = null

    init {
        tts = TextToSpeech(context.applicationContext, this)
    }

    override fun onInit(status: Int) {
        if (status == TextToSpeech.SUCCESS) {
            _isReady.value = true
            setupProgressListener()
            refreshAvailableVoices()
        } else {
            _isReady.value = false
        }
    }

    private fun setupProgressListener() {
        tts?.setOnUtteranceProgressListener(object : UtteranceProgressListener() {
            override fun onStart(utteranceId: String?) {
                _isSpeaking.value = true
            }

            override fun onDone(utteranceId: String?) {
                _isSpeaking.value = false
                onSpeechDoneCallback?.invoke()
            }

            @Deprecated("Deprecated in Java")
            override fun onError(utteranceId: String?) {
                _isSpeaking.value = false
            }

            override fun onError(utteranceId: String?, errorCode: Int) {
                _isSpeaking.value = false
            }
        })
    }

    fun refreshAvailableVoices() {
        val ttsInstance = tts ?: return
        val voiceList = mutableListOf<NovaVoiceOption>()

        try {
            val installedVoices = ttsInstance.voices
            if (!installedVoices.isNullOrEmpty()) {
                for (voice in installedVoices) {
                    val loc = voice.locale
                    val name = voice.name
                    val isFemale = name.contains("female", ignoreCase = true) || name.contains("-f-", ignoreCase = true)
                    val label = when {
                        loc.language == "bn" -> if (isFemale) "Google Female (bn-BD)" else "Google Male (bn-BD)"
                        loc.language == "hi" -> if (isFemale) "Google Female (hi-IN)" else "Google Male (hi-IN)"
                        loc.language == "en" -> if (isFemale) "Google Female (en-US)" else "Google Male (en-US)"
                        loc.language == "es" -> if (isFemale) "Google Female (es-ES)" else "Google Male (es-ES)"
                        loc.language == "ar" -> if (isFemale) "Google Female (ar-SA)" else "Google Male (ar-SA)"
                        else -> voice.name
                    }
                    voiceList.add(NovaVoiceOption(id = voice.name, displayName = label, localeTag = loc.toLanguageTag(), isFemale = isFemale))
                }
            }
        } catch (_: Exception) {}

        // Fallback default options if device list is empty or restricted
        if (voiceList.isEmpty()) {
            voiceList.addAll(
                listOf(
                    NovaVoiceOption("en_us_female", "Google Female (en-US)", "en-US", true),
                    NovaVoiceOption("en_us_male", "Google Male (en-US)", "en-US", false),
                    NovaVoiceOption("bn_bd_female", "Google Female (bn-BD)", "bn-BD", true),
                    NovaVoiceOption("bn_bd_male", "Google Male (bn-BD)", "bn-BD", false),
                    NovaVoiceOption("hi_in_female", "Google Female (hi-IN)", "hi-IN", true)
                )
            )
        }

        // Distinct by displayName to keep UI neat
        _availableVoices.value = voiceList.distinctBy { it.displayName }
    }

    fun speak(
        text: String,
        languageCode: String,
        speechRate: Float = 1.0f,
        voiceName: String? = null,
        onDone: (() -> Unit)? = null
    ) {
        val ttsInstance = tts ?: return
        if (!_isReady.value || text.isBlank()) return

        onSpeechDoneCallback = onDone

        val locale = when (languageCode.lowercase()) {
            "bn", "bengali" -> Locale.forLanguageTag("bn-BD")
            "hi", "hindi" -> Locale.forLanguageTag("hi-IN")
            "ar", "arabic" -> Locale.forLanguageTag("ar-SA")
            "es", "spanish" -> Locale.forLanguageTag("es-ES")
            else -> Locale.US
        }

        ttsInstance.language = locale
        ttsInstance.setSpeechRate(speechRate.coerceIn(0.5f, 2.0f))

        // Set voice if user specified and available
        if (!voiceName.isNullOrBlank()) {
            try {
                val matched = ttsInstance.voices?.firstOrNull { it.name == voiceName }
                if (matched != null) {
                    ttsInstance.voice = matched
                }
            } catch (_: Exception) {}
        }

        // Filter out markdown formatting asterisks/hashes for smoother spoken speech
        val spokenText = text
            .replace(Regex("[*#_`~>•]"), " ")
            .replace(Regex("\\s+"), " ")
            .trim()

        val params = Bundle().apply {
            putString(TextToSpeech.Engine.KEY_PARAM_UTTERANCE_ID, "nova_ai_utterance_${System.currentTimeMillis()}")
        }

        ttsInstance.speak(spokenText, TextToSpeech.QUEUE_FLUSH, params, "nova_speech_id")
        _isSpeaking.value = true
    }

    fun stop() {
        try {
            tts?.stop()
        } catch (_: Exception) {}
        _isSpeaking.value = false
    }

    fun shutdown() {
        try {
            tts?.stop()
            tts?.shutdown()
        } catch (_: Exception) {}
        tts = null
        _isReady.value = false
        _isSpeaking.value = false
    }
}

@Composable
fun NovaOrb(
    state: NovaVoiceState,
    modifier: Modifier = Modifier,
    size: Dp = 190.dp,
    audioLevel: Float = 0f,
    onClick: (() -> Unit)? = null
) {
    val infiniteTransition = rememberInfiniteTransition(label = "NovaOrbAnimation")

    val pulseScale by infiniteTransition.animateFloat(
        initialValue = 0.95f,
        targetValue = 1.05f,
        animationSpec = infiniteRepeatable(
            animation = tween(durationMillis = if (state == NovaVoiceState.LISTENING) 700 else 1800, easing = LinearEasing),
            repeatMode = RepeatMode.Reverse
        ),
        label = "OrbPulse"
    )

    val rotationAngle by infiniteTransition.animateFloat(
        initialValue = 0f,
        targetValue = 360f,
        animationSpec = infiniteRepeatable(
            animation = tween(durationMillis = if (state == NovaVoiceState.PROCESSING) 2000 else 8000, easing = LinearEasing),
            repeatMode = RepeatMode.Restart
        ),
        label = "OrbRotation"
    )

    val wavePhase by infiniteTransition.animateFloat(
        initialValue = 0f,
        targetValue = 6.28318f,
        animationSpec = infiniteRepeatable(
            animation = tween(durationMillis = if (state == NovaVoiceState.SPEAKING) 900 else 2400, easing = LinearEasing),
            repeatMode = RepeatMode.Restart
        ),
        label = "WavePhase"
    )

    val interactionSource = remember { MutableInteractionSource() }

    Box(
        modifier = modifier
            .size(size)
            .then(
                if (onClick != null) {
                    Modifier.clickable(
                        interactionSource = interactionSource,
                        indication = null,
                        onClick = onClick
                    )
                } else Modifier
            ),
        contentAlignment = Alignment.Center
    ) {
        Canvas(modifier = Modifier.size(size)) {
            val center = Offset(this.size.width / 2f, this.size.height / 2f)
            val baseRadius = this.size.minDimension / 2f * 0.72f * pulseScale

            val (outerGlowColor, ringGradient, centerSoundColor) = when (state) {
                NovaVoiceState.IDLE -> Triple(
                    NovaPurpleLight.copy(alpha = 0.35f),
                    listOf(NovaElectricPurple, NovaVividBlue, NovaCyanGlow, NovaElectricPurple),
                    NovaCyanBright
                )
                NovaVoiceState.LISTENING -> Triple(
                    NovaCyanGlow.copy(alpha = 0.55f),
                    listOf(NovaCyanBright, NovaCyanGlow, NovaVividBlue, NovaCyanBright),
                    Color.White
                )
                NovaVoiceState.PROCESSING -> Triple(
                    NovaElectricPurple.copy(alpha = 0.5f),
                    listOf(NovaPurpleLight, NovaElectricPurple, NovaVividBlue, NovaCyanGlow),
                    NovaCyanGlow
                )
                NovaVoiceState.SPEAKING -> Triple(
                    NovaElectricPurple.copy(alpha = 0.55f),
                    listOf(NovaPurpleLight, NovaElectricPurple, NovaCyanBright, NovaPurpleLight),
                    Color.White
                )
                NovaVoiceState.ERROR -> Triple(
                    NovaError.copy(alpha = 0.45f),
                    listOf(NovaError, Color(0xFFFB7185), NovaError),
                    Color.White
                )
            }

            // 1. Soft Outermost Atmospheric Radial Glow
            drawCircle(
                brush = Brush.radialGradient(
                    colors = listOf(outerGlowColor, outerGlowColor.copy(alpha = 0.08f), Color.Transparent),
                    center = center,
                    radius = baseRadius * 1.35f
                ),
                radius = baseRadius * 1.35f,
                center = center
            )

            // 2. Secondary Ring Pulse for listening / speaking
            if (state == NovaVoiceState.LISTENING || state == NovaVoiceState.SPEAKING) {
                val dynamicRingRadius = baseRadius * (1.12f + audioLevel * 0.22f)
                drawCircle(
                    color = outerGlowColor.copy(alpha = 0.35f),
                    radius = dynamicRingRadius,
                    center = center,
                    style = Stroke(width = 3.dp.toPx())
                )
            }

            // 3. Rotating Gradient Outer Orbit
            rotate(rotationAngle, pivot = center) {
                drawCircle(
                    brush = Brush.sweepGradient(ringGradient, center = center),
                    radius = baseRadius,
                    center = center,
                    style = Stroke(width = 6.dp.toPx())
                )
            }

            // 4. Inner Dark Spherical Core
            drawCircle(
                brush = Brush.radialGradient(
                    colors = listOf(
                        Color(0xFF0F172A),
                        Color(0xFF070B16),
                        Color(0xFF03050B)
                    ),
                    center = center,
                    radius = baseRadius * 0.88f
                ),
                radius = baseRadius * 0.88f,
                center = center
            )

            // 5. Delicate Inner Neon Border
            drawCircle(
                color = outerGlowColor.copy(alpha = 0.6f),
                radius = baseRadius * 0.86f,
                center = center,
                style = Stroke(width = 1.5.dp.toPx())
            )

            // 6. Central Sound Waveform Bars (reproducing the iconic 5 audio bars in the reference image)
            val barCount = 5
            val maxBarHeight = baseRadius * 0.65f
            val barSpacing = 7.dp.toPx()
            val totalWidth = (barCount - 1) * barSpacing
            val startX = center.x - (totalWidth / 2f)

            for (i in 0 until barCount) {
                val x = startX + i * barSpacing
                val normalizedIdx = (i - 2).toFloat() // -2, -1, 0, 1, 2
                val baseFactor = 1f - (kotlin.math.abs(normalizedIdx) * 0.22f)

                val dynamicHeight = when (state) {
                    NovaVoiceState.IDLE -> {
                        val sine = sin(wavePhase + i * 0.9f)
                        maxBarHeight * (0.35f + 0.18f * sine) * baseFactor
                    }
                    NovaVoiceState.LISTENING -> {
                        val levelMultiplier = (0.35f + audioLevel * 0.95f).coerceIn(0.2f, 1.3f)
                        val sine = sin(wavePhase * 2.5f + i * 1.2f)
                        maxBarHeight * levelMultiplier * (0.5f + 0.5f * kotlin.math.abs(sine))
                    }
                    NovaVoiceState.PROCESSING -> {
                        val sine = sin(wavePhase * 1.5f + i * 0.8f)
                        maxBarHeight * (0.4f + 0.25f * sine) * baseFactor
                    }
                    NovaVoiceState.SPEAKING -> {
                        val sine = sin(wavePhase * 3.2f + i * 1.4f)
                        maxBarHeight * (0.5f + 0.5f * kotlin.math.abs(sine)) * baseFactor
                    }
                    NovaVoiceState.ERROR -> {
                        maxBarHeight * 0.3f * baseFactor
                    }
                }.coerceAtLeast(6.dp.toPx())

                drawLine(
                    color = centerSoundColor,
                    start = Offset(x, center.y - dynamicHeight / 2f),
                    end = Offset(x, center.y + dynamicHeight / 2f),
                    strokeWidth = 4.dp.toPx(),
                    cap = StrokeCap.Round
                )
            }
        }
    }

@Composable
fun WaveformVisualizer(
    modifier: Modifier = Modifier,
    barCount: Int = 28,
    height: Dp = 64.dp,
    isActive: Boolean = true,
    isSpeakingMode: Boolean = false,
    audioLevel: Float = 0f
) {
    val infiniteTransition = rememberInfiniteTransition(label = "WaveformAnim")
    val phase by infiniteTransition.animateFloat(
        initialValue = 0f,
        targetValue = 6.28318f,
        animationSpec = infiniteRepeatable(
            animation = tween(durationMillis = if (isSpeakingMode) 900 else 1200, easing = LinearEasing),
            repeatMode = RepeatMode.Restart
        ),
        label = "WaveformPhase"
    )

    val gradientColors = if (isSpeakingMode) {
        listOf(NovaPurpleLight, NovaElectricPurple, NovaCyanBright, NovaPurpleLight)
    } else {
        listOf(NovaCyanGlow, NovaCyanBright, NovaElectricPurple, NovaCyanGlow)
    }

    Canvas(
        modifier = modifier
            .fillMaxWidth()
            .height(height)
    ) {
        val centerY = size.height / 2f
        val step = size.width / (barCount + 1)
        val barStroke = (step * 0.45f).coerceIn(2.5.dp.toPx(), 4.5.dp.toPx())

        for (i in 0 until barCount) {
            val x = step * (i + 1)

            // Bell curve shape from edges towards center
            val normalizedDistFromCenter = kotlin.math.abs(i - barCount / 2f) / (barCount / 2f)
            val bellFactor = (1f - normalizedDistFromCenter * 0.7f).coerceIn(0.2f, 1f)

            val currentHeight = if (isActive) {
                val sine1 = sin(phase * 1.8f + i * 0.45f)
                val sine2 = sin(phase * 3.1f + i * 0.85f)
                val combinedSine = ((sine1 + sine2) / 2f) * 0.5f + 0.5f

                val levelBoost = if (!isSpeakingMode) audioLevel * 0.6f else 0.3f
                size.height * bellFactor * (0.25f + combinedSine * 0.5f + levelBoost)
            } else {
                size.height * 0.12f * bellFactor
            }.coerceIn(4.dp.toPx(), size.height * 0.95f)

            val brush = Brush.verticalGradient(
                colors = gradientColors,
                startY = centerY - currentHeight / 2f,
                endY = centerY + currentHeight / 2f
            )

            drawLine(
                brush = brush,
                start = Offset(x, centerY - currentHeight / 2f),
                end = Offset(x, centerY + currentHeight / 2f),
                strokeWidth = barStroke,
                cap = StrokeCap.Round
            )
        }
    }
}

    onNavigateToSettings: () -> Unit,
    onSelectConversation: (ConversationEntity) -> Unit
) {
    val conversations by viewModel.allConversations.collectAsState()
    var showClearConfirm by remember { mutableStateOf(false) }

    Scaffold(
        containerColor = NovaDarkBg,
        topBar = {
            NovaTopBar(
                title = "Conversation History",
                showBack = false,
                actions = {
                    if (conversations.isNotEmpty()) {
                        TextButton(
                            onClick = { showClearConfirm = true },
                            modifier = Modifier.testTag("history_clear_all_button")
                        ) {
                            Text(
                                text = "Clear All",
                                fontSize = 14.sp,
                                color = NovaError,
                                fontWeight = FontWeight.SemiBold
                            )
                        }
                    }
                }
            )
        },
        bottomBar = {
            NovaBottomBar(
                currentTab = NovaNavTab.HISTORY,
                onTabSelected = { tab ->
                    when (tab) {
                        NovaNavTab.HOME -> onNavigateToHome()
                        NovaNavTab.HISTORY -> {}
                        NovaNavTab.SETTINGS -> onNavigateToSettings()
                    }
                }
            )
        }
    ) { innerPadding ->
        Box(
            modifier = Modifier
                .fillMaxSize()
                .padding(innerPadding)
                .padding(horizontal = 20.dp)
        ) {
            if (conversations.isEmpty()) {
                // Empty state illustration
                Column(
                    modifier = Modifier.fillMaxSize(),
                    horizontalAlignment = Alignment.CenterHorizontally,
                    verticalArrangement = Arrangement.Center
                ) {
                    Box(
                        modifier = Modifier
                            .size(100.dp)
                            .clip(CircleShape)
                            .background(NovaDarkSurface)
                            .border(1.dp, NovaDarkBorder, CircleShape),
                        contentAlignment = Alignment.Center
                    ) {
                        Icon(
                            imageVector = Icons.Default.History,
                            contentDescription = "No History",
                            tint = NovaTextMuted,
                            modifier = Modifier.size(46.dp)
                        )
                    }

                    Spacer(modifier = Modifier.height(20.dp))

                    Text(
                        text = "No Conversations Yet",
                        fontSize = 18.sp,
                        fontWeight = FontWeight.Bold,
                        color = NovaTextPrimary
                    )

                    Spacer(modifier = Modifier.height(8.dp))

                    Text(
                        text = "Your voice queries and responses will be saved here automatically.",
                        fontSize = 13.sp,
                        color = NovaTextMuted,
                        textAlign = TextAlign.Center,
                        modifier = Modifier.padding(horizontal = 32.dp)
                    )
                }
            } else {
                LazyColumn(
                    modifier = Modifier.fillMaxSize(),
                    verticalArrangement = Arrangement.spacedBy(10.dp)
                ) {
                    items(conversations, key = { it.id }) { item ->
                        ConversationHistoryItem(
                            conversation = item,
                            onClick = { onSelectConversation(item) },
                            onDelete = { viewModel.deleteConversation(item.id) }
                        )
                    }
                    item {
                        Spacer(modifier = Modifier.height(16.dp))
                    }
                }
            }
        }
    }

    if (showClearConfirm) {
        AlertDialog(
            onDismissRequest = { showClearConfirm = false },
            title = {
                Text("Clear History", color = NovaTextPrimary, fontWeight = FontWeight.Bold)

            text = {

                    Text("Cancel", color = NovaTextSecondary)
                }
            },
            containerColor = NovaDarkSurfaceVariant,
            shape = RoundedCornerShape(20.dp)
        )

        Spacer(modifier = Modifier.width(14.dp))

            shape = RoundedCornerShape(20.dp)
        )
    }

    if (showTermsDialog) {
        AlertDialog(
            onDismissRequest = { showTermsDialog = false },
            title = { Text("Terms of Service", color = NovaTextPrimary, fontWeight = FontWeight.Bold) },
            text = {
                Text(
                    text = "By using NOVA AI, you agree to:\n\n" +
                            "• Use the personal voice assistant responsibly and ethically.\n" +
                            "• Abide by Google Generative AI terms and rate limits.\n" +
                            "• Understand that AI responses are generative and should be verified for critical matters.",
                    color = NovaTextSecondary,
                    fontSize = 13.sp,
                    lineHeight = 20.sp
                )
            },
            confirmButton = {
                TextButton(onClick = { showTermsDialog = false }) {
                    Text("Close", color = NovaCyanBright)
                }
            },
            containerColor = NovaDarkSurfaceVariant,
            shape = RoundedCornerShape(20.dp)
        )
    }
}

        Text(
            text = label,

        )
    }
